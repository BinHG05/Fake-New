class CrossAttentionFusion(nn.Module):
    """
    Cross-attention: text embedding attends to image embedding.

    Architecture:
      text (768d) → Q projection (fusion_dim)
      image (512d) → K, V projection (fusion_dim)
      attended = MultiHeadAttention(Q=text, K=image, V=image)
      gate = σ(W · [text_proj; attended])
      output = gate * attended + (1-gate) * text_proj
    """
    def __init__(self, text_dim=768, image_dim=512,
                 fusion_dim=256, num_heads=4, dropout=0.1):
        super().__init__()
        # Project modalities to shared space
        self.text_proj = nn.Linear(text_dim, fusion_dim)
        self.image_proj = nn.Linear(image_dim, fusion_dim)

        # Multi-head attention (text as Q, image as K/V)
        self.attn = nn.MultiheadAttention(
            embed_dim=fusion_dim, num_heads=num_heads,
            dropout=dropout, batch_first=True,
        )
        # Gated residual
        self.gate_linear = nn.Linear(fusion_dim * 2, 1)
        self.norm = nn.LayerNorm(fusion_dim)

    def forward(self, text_emb, image_emb):
        t = self.text_proj(text_emb)     # [B, fusion_dim]
        i = self.image_proj(image_emb)   # [B, fusion_dim]
        t_seq = t.unsqueeze(1)
        i_seq = i.unsqueeze(1)

        # Cross attention: text queries image
        attended, _ = self.attn(query=t_seq, key=i_seq, value=i_seq)
        attended = attended.squeeze(1)

        # Gated residual: learn to blend text vs attended
        gate = torch.sigmoid(
            self.gate_linear(torch.cat([t, attended], dim=-1))
        )
        fused = gate * attended + (1 - gate) * t
        return self.norm(fused)

from torch_geometric.nn import SAGEConv

class MultimodalGraphNet(nn.Module):
    """
    End-to-end multimodal GNN.
    Takes raw text + image + graph structure → 6-class logits.
    """
    def __init__(self, fusion_dim=256, hidden_dim=256,
                 num_classes=6, num_gnn_layers=2, dropout=0.3):
        super().__init__()
        # --- GNN layers (GraphSAGE) ---
        self.gnn_layers = nn.ModuleList()
        self.gnn_norms = nn.ModuleList()
        for i in range(num_gnn_layers):
            in_dim = fusion_dim if i == 0 else hidden_dim
            self.gnn_layers.append(SAGEConv(in_dim, hidden_dim))
            self.gnn_norms.append(nn.LayerNorm(hidden_dim))

    def forward(self, h, edge_index):
        # GNN message passing with residual connections
        for gnn, norm in zip(self.gnn_layers, self.gnn_norms):
            h_new = gnn(h, edge_index)
            h_new = norm(h_new)
            h_new = F.relu(h_new)
            h_new = F.dropout(h_new, p=self.dropout,
                              training=self.training)
            # Residual connection
            if h.shape == h_new.shape:
                h = h + h_new
            else:
                h = h_new
        return self.classifier(h)

from transformers import AutoModel

class TextOnlyModel(nn.Module):
    """
    Text-only baseline using frozen XLM-RoBERTa.
    Architecture:
      xlm-roberta-base (frozen) → [CLS] token (768d)
      → Linear(768, num_classes)
    """
    def __init__(self, num_classes=6,
                 backbone='xlm-roberta-base'):
        super().__init__()
        self.backbone = AutoModel.from_pretrained(backbone)
        # Freeze backbone — chỉ train lớp classifier
        for param in self.backbone.parameters():
            param.requires_grad = False

        hidden_size = self.backbone.config.hidden_size  # 768
        self.classifier = nn.Sequential(
            nn.Dropout(0.3),
            nn.Linear(hidden_size, num_classes),
        )

    def forward(self, input_ids, attention_mask):
        with torch.no_grad():
            outputs = self.backbone(
                input_ids=input_ids,
                attention_mask=attention_mask
            )
        # Lấy [CLS] token (vị trí 0)
        cls_emb = outputs.last_hidden_state[:, 0, :]
        return self.classifier(cls_emb)

from torchvision import models

class ImageOnlyModel(nn.Module):
    """Frozen ResNet-50 + Linear Classifier."""
    def __init__(self, num_classes=6):
        super().__init__()
        resnet = models.resnet50(
            weights=models.ResNet50_Weights.IMAGENET1K_V1
        )
        # Lấy tất cả layers trừ fc cuối
        self.features = nn.Sequential(
            *list(resnet.children())[:-1]
        )  # output: [B, 2048, 1, 1]

        # Freeze backbone
        for param in self.features.parameters():
            param.requires_grad = False

        self.classifier = nn.Sequential(
            nn.Dropout(0.3),
            nn.Linear(2048, num_classes),
        )

    def forward(self, images):
        with torch.no_grad():
            features = self.features(images)  # [B, 2048, 1, 1]
        features = features.flatten(1)  # [B, 2048]
        return self.classifier(features)

# x = [N, 1280] (768 text_emb + 512 image_emb từ CLIP)

class PrecomputedMultimodalGNN(nn.Module):
    """
    Variant works with PRE-COMPUTED embeddings.
    x = [text_emb || image_emb] from CLIP.
    """
    def forward(self, x, edge_index):
        # Split precomputed CLIP features
        text_emb = x[:, :self.text_dim]   # [N, 768]
        image_emb = x[:, self.text_dim:]  # [N, 512]

        # Fusion
        h = self.fusion(text_emb, image_emb)
        # GNN message passing...
        return self.classifier(h)

from sklearn.model_selection import train_test_split

def split_class(class_records):
    """Stratified split using sklearn"""
    if len(class_records) < 3:
        return class_records, [], []

    train, temp = train_test_split(
        class_records,
        test_size=(1 - self.train_ratio),
        random_state=42
    )
    if len(temp) >= 2:
        val_ratio_adjusted = (
            self.val_ratio / (1 - self.train_ratio)
        )
        val, test = train_test_split(
            temp,
            test_size=(1 - val_ratio_adjusted),
            random_state=42
        )
    else:
        val, test = temp, []
    return train, val, test

from PIL import Image, ImageFile, ImageOps

def process_image(self, url, post_id, use_padding=True):
    """Download and process image"""
    image = self._download_image(url)
    # Convert to RGB if needed
    if image.mode != 'RGB':
        image = image.convert('RGB')

    if use_padding:
        # Resize keeping aspect ratio
        image.thumbnail(self.target_size,
                        Image.Resampling.LANCZOS)
        # Create new image with padding color
        new_image = Image.new('RGB', self.target_size,
                              self.padding_color)
        # Paste resized image in center
        paste_x = (self.target_size[0] - image.width) // 2
        paste_y = (self.target_size[1] - image.height) // 2
        new_image.paste(image, (paste_x, paste_y))
    else:
        # Center crop
        processed_image = ImageOps.fit(
            image, self.target_size,
            method=Image.Resampling.LANCZOS
        )
    processed_image.save(save_path, "JPEG", quality=85)

    """Start a training run via REST API."""
    data = request.json or {}
    model_type = data.get("model_type", "text")
    epochs = data.get("epochs", 20)
    lr = data.get("lr", 2e-4)

    if model_type in ("text", "image", "fusion"):
        script = os.path.join(PROJECT_ROOT,
            "src", "training", "train_baseline.py")

                "--epochs", str(epochs),

        script = os.path.join(PROJECT_ROOT,
            "src", "training", "train_multimodal_gnn.py")
        fusion_type = data.get("fusion_type",
                               "cross_attention")

    task_id = launch_script(task_type, script, args)
    return jsonify({"task_id": task_id, "status": "started"})
